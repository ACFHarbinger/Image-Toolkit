def T(f,n=1):
    t=time.perf_counter()
    for _ in range(n): out=f()
    return (time.perf_counter()-t)/n*1000, out

# Gemini: exact BLAS identity
def blas_idx():
    rgb=scaled[...,:3].astype(np.float32)
    h,w=rgb.shape[:2]
    yy,xx=np.mgrid[0:h,0:w]
    thr=(enc._BAYER_8[yy%8,xx%8]+0.5)/64.0-0.5
    adj=(rgb+thr[:,:,None]*24.0).reshape(-1,3)
    half=0.5*(pal**2).sum(1)
    out=np.empty(adj.shape[0],np.uint8)
    for s in range(0,adj.shape[0],65536):
        c=adj[s:s+65536]; out[s:s+65536]=(half[None,:]-c@pal.T).argmin(1)
    return out.reshape(h,w)
ms,idx2=T(blas_idx,1); print("BLAS-identity index: %.0f ms/frame, identical to current: %s, mismatches %d"%(ms,np.array_equal(idx,idx2),int((idx!=idx2).sum())))
# Gemini: 7-bit LUT
def lut_idx():
    g=np.arange(128,dtype=np.float32)*(255/127)
    R,G,B=np.meshgrid(g,g,g,indexing="ij"); pts=np.stack([R,G,B],-1).reshape(-1,3)
    half=0.5*(pal**2).sum(1); lut=np.empty(pts.shape[0],np.uint8)
    for s in range(0,pts.shape[0],65536):
        lut[s:s+65536]=(half[None,:]-pts[s:s+65536]@pal.T).argmin(1)
    return lut
t=time.perf_counter(); lut=lut_idx(); build=(time.perf_counter()-t)*1000

def dense(idx_,w_,s):
    dst=idx_.shape[1]; M=np.zeros((dst,s),np.float32); 
    np.add.at(M,(np.tile(np.arange(dst),idx_.shape[0]),idx_.reshape(-1)),w_.reshape(-1)); return M

def sgemm():
    out=np.empty((dH,dW,3),np.float32)
    for c in range(3): out[...,c]=Wy@(np.ascontiguousarray(src[...,c])@Wx.T)
    return out
ms,o=T(sgemm,3); print("sgemm separable (3ch): %.1f ms/frame"%ms)
# clamp: 6x6 min/max on source, gather by floor(src_pos)
def clamp(o):
    p=cv2.copyMakeBorder(src,2,3,2,3,cv2.BORDER_REPLICATE)
    kern=np.ones((6,6),np.uint8)
    lo=cv2.erode(p,kern,anchor=(2,2)); hi=cv2.dilate(p,kern,anchor=(2,2))
    # window for dst pixel starts at floor(pos)-2 .. floor(pos)+3 ; in padded coords offset +2 -> anchor at floor(pos)
    sx=((np.arange(dW)+0.5)*(W/dW)-0.5); sy=((np.arange(dH)+0.5)*(H/dH)-0.5)
    mx=np.tile(np.floor(sx).astype(np.float32)[None,:]+2,(dH,1)); my=np.tile(np.floor(sy).astype(np.float32)[:,None]+2,(1,dW))
    L=cv2.remap(lo,mx,my,cv2.INTER_NEAREST); Hh=cv2.remap(hi,mx,my,cv2.INTER_NEAREST)
    return np.clip(o,L,Hh)
